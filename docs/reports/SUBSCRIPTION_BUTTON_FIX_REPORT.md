**ПОСЛЕ (решение):**
```typescript
// ✅ ДОБАВЛЯЕМ ОБРАБОТЧИК КНОПКИ "ОФОРМИТЬ ПОДПИСКУ"
// Кнопка "💫 Оформить подписку" использует Markup.button.text (keyboard button),
// поэтому нужен hears обработчик
bot.hears(['💫 Оформить подписку', '💫 Subscribe'], async (ctx: MyContext) => {
  logger.info('GLOBAL HEARS: Subscribe button pressed', {
    telegramId: ctx.from?.id,
    text: ctx.message && 'text' in ctx.message ? ctx.message.text : 'unknown',
  })

  try {
    const is_ru = isRussianFromState(ctx)

    // Покидаем текущую сцену
    if (ctx.scene.current) {
      await ctx.scene.leave()
    }

    // Устанавливаем режим подписки и переходим в сцену
    ctx.session.mode = ModeEnum.SubscriptionScene
    await ctx.scene.enter(ModeEnum.SubscriptionScene)

    logger.info('✅ GLOBAL HEARS: Successfully entered subscription scene', {
      telegramId: ctx.from?.id,
    })
  } catch (error) {
    logger.error('❌ GLOBAL HEARS: Error in subscription handler:', {
      error: error instanceof Error ? error.message : String(error),
      telegramId: ctx.from?.id,
    })

    const isRuError = isRussianFromState(ctx)
    await ctx.reply(
      isRuError
        ? '❌ Произошла ошибка при переходе к оформлению подписки.'
        : '❌ An error occurred while accessing subscription.'
    )
  }
})

// ОБРАБОТЧИК ДЛЯ УВЕЛИЧЕНИЯ КАЧЕСТВА НЕЙРОФОТО...
```
